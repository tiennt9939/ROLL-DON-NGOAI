// ADD-ONLY RULE: chèn bên trong match /databases/{database}/documents
// và TRƯỚC match /{document=**}. Không thay toàn bộ Rules hiện hành.
// Catalog được website đọc qua match /apps/{appId} hiện có; chỉ đọc document.
// Không cấp quyền ghi vào apps, packingOrders, inventory, payroll hoặc các collection cũ.

match /externalOrders/{orderId} {
  allow read: if isAdmin() || (isActiveUser() && role() == 'manager');
  allow create: if (isAdmin() || (isActiveUser() && role() == 'manager'))
    && request.resource.data.source == 'ROLL_DON_NGOAI'
    && request.resource.data.createdByUid == request.auth.uid
    && request.resource.data.parcelCount is int
    && request.resource.data.parcelCount >= 1
    && request.resource.data.parcelCount <= 99
    && request.resource.data.items is list
    && request.resource.data.items.size() > 0
    && request.resource.data.total is number
    && request.resource.data.total >= 0;
  // Phiên bản hiện tại chỉ tạo, đọc và in lại; chưa cho sửa/xóa đơn.
  allow update, delete: if false;
}
