// CHỈ THÊM block này vào Firestore Rules hiện hành,
// bên trong match /databases/{database}/documents và TRƯỚC match /{document=**}.
// Giữ nguyên block /externalOrders hiện có và mọi Rules khác.

match /externalCatalog/{plantId} {
  // User đang hoạt động và admin được đọc danh mục để tạo đơn.
  allow read: if isAdmin() || isActiveUser();

  // Chỉ admin được thêm/sửa/xóa danh mục riêng của website.
  allow create, update: if isAdmin()
    && request.resource.data.source == 'ROLL_DON_NGOAI'
    && request.resource.data.name is string
    && request.resource.data.name.size() > 0
    && request.resource.data.sizes is list
    && request.resource.data.sizes.size() > 0;

  allow delete: if isAdmin();
}

// Các hàm isAdmin(), isActiveUser() phải là hàm đang có trong Rules hiện tại.
// Không thay đổi quyền của apps, packingOrders, inventory, payroll hoặc externalOrders.
